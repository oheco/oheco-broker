cmake_minimum_required(VERSION 3.16)
project(oheco_broker_c LANGUAGES C)
find_package(Threads REQUIRED)
add_library(oheco_broker_c OBJECT oheco_broker.c)
set_target_properties(oheco_broker_c PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(oheco_broker_c PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(oheco_broker_c PUBLIC c_std_11)
target_link_libraries(oheco_broker_c PUBLIC Threads::Threads)
option(OHECO_BROKER_REMOTE "Build the authenticated remote peer SDK using explicit dependency prefixes" OFF)
if(OHECO_BROKER_REMOTE)
  add_subdirectory(remote)
endif()
option(OHECO_BROKER_C_SMOKE "Build an explicitly supplied external-service C smoke executable" OFF)
set(OHECO_BROKER_C_SMOKE_SOURCE "" CACHE FILEPATH "Smoke source from the full repository checkout")
if(OHECO_BROKER_C_SMOKE)
  if(NOT EXISTS "${OHECO_BROKER_C_SMOKE_SOURCE}")
    message(FATAL_ERROR "Set OHECO_BROKER_C_SMOKE_SOURCE to the full-checkout example, or leave OHECO_BROKER_C_SMOKE OFF")
  endif()
  add_executable(oheco-broker-c-smoke "${OHECO_BROKER_C_SMOKE_SOURCE}")
  target_link_libraries(oheco-broker-c-smoke PRIVATE oheco_broker_c)
endif()
