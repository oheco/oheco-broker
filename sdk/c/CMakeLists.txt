cmake_minimum_required(VERSION 3.16)
project(oheco_broker_c LANGUAGES C)
find_package(Threads REQUIRED)
add_library(oheco_broker_c OBJECT oheco_broker.c)
set_target_properties(oheco_broker_c PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(oheco_broker_c PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(oheco_broker_c PUBLIC c_std_11)
target_link_libraries(oheco_broker_c PUBLIC Threads::Threads)
option(OHECO_BROKER_C_SMOKE "Build the external-service C smoke executable" OFF)
if(OHECO_BROKER_C_SMOKE)
  add_executable(oheco-broker-c-smoke ../../examples/c/smoke.c)
  target_link_libraries(oheco-broker-c-smoke PRIVATE oheco_broker_c)
endif()
