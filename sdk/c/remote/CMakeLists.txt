cmake_minimum_required(VERSION 3.16)
project(ob_remote C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(OB_NATIVE_PREFIX "" CACHE PATH "Explicit prefix of the pinned native dependencies")
set(OB_CURL_PREFIX "" CACHE PATH "Explicit prefix of curl built with the same BoringSSL")
if(NOT OB_NATIVE_PREFIX OR NOT OB_CURL_PREFIX)
  message(FATAL_ERROR "Set OB_NATIVE_PREFIX and OB_CURL_PREFIX, or use the SDK's offline build.sh first")
endif()
find_package(Threads REQUIRED)
find_library(OB_xquic_LIB NAMES xquic-static xquic PATHS "${OB_NATIVE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
foreach(dep juice cjson ssl crypto)
  find_library(OB_${dep}_LIB NAMES ${dep} PATHS "${OB_NATIVE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
find_library(OB_curl_LIB NAMES curl PATHS "${OB_CURL_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
# These native static archives include BoringSSL C++. Select the compiler's
# native runtime; callers may choose explicitly without modifying SDK sources.
execute_process(COMMAND "${CMAKE_C_COMPILER}" -dumpmachine OUTPUT_VARIABLE _ob_triple OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE _ob_compiler_status)
if(NOT _ob_compiler_status EQUAL 0)
  message(FATAL_ERROR "Cannot identify the native C compiler target")
endif()
if(_ob_triple MATCHES "ohos")
  set(_ob_runtime c++)
else()
  set(_ob_runtime stdc++)
endif()
set(OB_CXX_RUNTIME "${_ob_runtime}" CACHE STRING "Native C++ runtime: c++ or stdc++")
if(NOT OB_CXX_RUNTIME MATCHES "^(c[+][+]|stdc[+][+])$")
  message(FATAL_ERROR "OB_CXX_RUNTIME must be c++ or stdc++")
endif()
add_library(ob_remote STATIC ob_remote.c ob_api.c ob_json.c ob_ws.c peer/crypto.c peer/control.c peer/engine.c peer/target.c peer/mapping.c)
set_target_properties(ob_remote PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(ob_remote PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${OB_NATIVE_PREFIX}/include" "${OB_CURL_PREFIX}/include")
target_compile_definitions(ob_remote PRIVATE _GNU_SOURCE JUICE_STATIC CURL_STATICLIB)
target_compile_options(ob_remote PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ob_remote PUBLIC ${OB_xquic_LIB} ${OB_juice_LIB} ${OB_cjson_LIB} ${OB_curl_LIB} ${OB_ssl_LIB} ${OB_crypto_LIB} ${OB_CXX_RUNTIME} Threads::Threads m ${CMAKE_DL_LIBS})
option(OB_REMOTE_BUILD_TESTS "Build full-checkout native peer tests" OFF)
set(OB_REMOTE_TEST_SOURCE_DIR "" CACHE PATH "Explicit directory containing the repository C tests")
if(OB_REMOTE_BUILD_TESTS)
  foreach(test remote_peer_test websocket_test)
    if(NOT EXISTS "${OB_REMOTE_TEST_SOURCE_DIR}/${test}.c")
      message(FATAL_ERROR "OB_REMOTE_BUILD_TESTS requires OB_REMOTE_TEST_SOURCE_DIR from a full checkout")
    endif()
    add_executable(${test} "${OB_REMOTE_TEST_SOURCE_DIR}/${test}.c")
    target_include_directories(${test} PRIVATE "${OB_NATIVE_PREFIX}/include" "${OB_CURL_PREFIX}/include")
    target_link_libraries(${test} PRIVATE ob_remote)
  endforeach()
endif()
install(TARGETS ob_remote ARCHIVE DESTINATION lib)
install(FILES ob_remote.h ob_api.h DESTINATION include/oheco-broker)
